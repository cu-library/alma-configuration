rule "add title from 245 and language code"
salience 80
when
  (true)
then
  copyField "245.a" to "988.t"
  suffixSubfield "988.t" with "245.b"
  replaceContents "988.t. /" with ""
  addField "988.e.eng"
end

rule "add oclc nums from 035/095 in record if it is an English description"
salience 70
when
  (exists "095.a.(OCoLC)*" or exists "035.a.(OCoLC)*") and exists "040.b.eng"
then
  copyField "095" to "988" if (exists "095.a.(OCoLC)*")
  copyField "035" to "988" if (exists "035.a.(OCoLC)*")
  changeSubField "988.a" to "w"
  changeSubField "988.z" to "w"
end

rule "otherwise add blank $w"
salience 70
when
  not exists "040.b.eng"
then
  addField "988.w.(OCoLC)"
end

rule "add $6 linkage in 035 if it is a French description"
salience 60
when
  exists "040.b.fre"
then
  addSubField "035.6.788-01" if (exists "035.a.(OCoLC)*")
end

rule "finally, convert temporary 988 field to 788"
salience 50
when
  (true)
then
  changeSubField "988.x" to "a"
  combineFields "988" excluding ""
  changeField "988" to "788"
end
